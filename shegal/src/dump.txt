// --- 1. Helper function to retrieve the csrftoken cookie ---
const getCsrfToken = () => {
  let cookieValue = null;
  if (document.cookie && document.cookie !== '') {
    const cookies = document.cookie.split(';');
    for (let i = 0; i < cookies.length; i++) {
      const cookie = cookies[i].trim();
      // Does this cookie string begin with the name we want?
      if (cookie.substring(0, 'csrftoken'.length + 1) === ('csrftoken' + '=')) {
        cookieValue = decodeURIComponent(cookie.substring('csrftoken'.length + 1));
        break;
      }
    }
  }
  return cookieValue;
};

const handleSubmit = async () => {
  if (!image) return alert("Please upload an image!");

  // Reset state before starting the request
  setLoadingg(true);
  setErrror(null);
  setAnalysisResult(null); 

  const accessToken = localStorage.getItem("access_token");
  const csrfToken = getCsrfToken(); 

  if (!accessToken) {
    setLoadingg(false);
    return console.error("Authentication Error: access_token is missing from localStorage.");
  }
  if (!csrfToken) {
    setLoadingg(false);
    return console.error("CSRF Error: csrftoken cookie is missing or unreadable.");
  }

  const formData = new FormData();
  formData.append("image", image);

  axios.defaults.withCredentials = true;

  try {
    const res = await axios.post(
      "http://localhost:8000/store/skin-analysis/",
      formData,
      {
        headers: {
          'Authorization': `Bearer ${accessToken}`,
          'X-CSRFToken': csrfToken, 
          'Content-Type': 'multipart/form-data', 
        },
      }
    );
    
    // **THIS IS THE CRITICAL CHANGE:** Store the data in state
    setAnalysisResult(res.data); 
    console.log("Analysis Successful:", res.data);

  } catch (err) {
    // Set the error state on failure
    if (err.response) {
      setErrror(`Error ${err.response.status}: ${err.response.data.detail || JSON.stringify(err.response.data)}`);
    } else if (err.request) {
      setErrror("Network Error: No response received from server.");
    } else {
      setErrror(`Request Setup Error: ${err.message}`);
    }
  } finally {
    setLoadingg(false); // Stop loading regardless of outcome
  }
